#include <asm/avx512fintrin.h>     // если AVX-512

#define DEVICE_NAME "fwht_accel"
#define IOCTL_FWHT _IOWR('f', 1, struct fwht_arg)

struct fwht_arg {
    void __user *in;
    void __user *out;
    size_t len;                 // в float32 элементах, кратно 256
};

static long device_ioctl(struct file *file, unsigned int cmd, unsigned long arg)
{
    struct fwht_arg ua;
    float *k_in = NULL, *k_out = NULL;
    int ret = -EINVAL;

    if (copy_from_user(&ua, (void *)arg, sizeof(ua)))
        return -EFAULT;

    if (ua.len % 256 != 0 || ua.len == 0)
        return -EINVAL;

    k_in  = kmalloc(ua.len * sizeof(float), GFP_KERNEL);
    k_out = kmalloc(ua.len * sizeof(float), GFP_KERNEL);
    if (!k_in || !k_out) goto out;

    if (copy_from_user(k_in, ua.in, ua.len * sizeof(float))) goto out;

    kernel_fpu_begin();

    // Здесь твой FWHT (можно взять из userspace-версии и адаптировать)
    // Простейший вариант — скопировать логику _fwht_f32 из numba, но на C + intrinsics
    // Для примера — заглушка
    for (size_t i = 0; i < ua.len; i++)
        k_out[i] = k_in[i] * 2.0f;  // ← заменить на реальный FWHT

    kernel_fpu_end();

    if (copy_to_user(ua.out, k_out, ua.len * sizeof(float)))
        ret = -EFAULT;
    else
        ret = 0;

out:
    kfree(k_out);
    kfree(k_in);
    return ret;
}

Шаг 6. fwht_avx512.c (самое важное — ядро FWHT)
C// fwht_avx512.c

#include <linux/kernel.h>
#include <linux/types.h>
#include <asm/fpu/api.h>
#include <asm/avx512fintrin.h>

#define N 256

static inline void fwht_avx512_block(float *y, int len) {
    // len должен быть степенью двойки, кратной 16 (AVX-512)
    for (int h = 1; h < len; h <<= 1) {
        int step = h << 1;
        for (int i = 0; i < len; i += step) {
            for (int j = 0; j < h; j += 16) {
                // загружаем 16 float за раз
                __m512 a = _mm512_loadu_ps(y + i + j);
                __m512 b = _mm512_loadu_ps(y + i + h + j);

                __m512 sum  = _mm512_add_ps(a, b);
                __m512 diff = _mm512_sub_ps(a, b);

                _mm512_storeu_ps(y + i + j,     sum);
                _mm512_storeu_ps(y + i + h + j, diff);
            }
        }
    }
}

void fwht_accel_single(float *y) {
    kernel_fpu_begin();

    fwht_avx512_block(y, N);

    // нормализация / sqrt(N)
    const __m512 inv_sqrt_n = _mm512_set1_ps(1.0f / sqrtf((float)N));
    for (int i = 0; i < N; i += 16) {
        __m512 v = _mm512_loadu_ps(y + i);
        v = _mm512_mul_ps(v, inv_sqrt_n);
        _mm512_storeu_ps(y + i, v);
    }

    kernel_fpu_end();
}

// batch-версия (пока последовательная, дальше можно распараллелить workqueue)
void fwht_accel_batch(float *in_out, uint32_t k) {
    kernel_fpu_begin();

    for (uint32_t row = 0; row < k; row++) {
        float *y = in_out + row * N;
        fwht_avx512_block(y, N);

        const __m512 inv = _mm512_set1_ps(1.0f / sqrtf((float)N));
        for (int i = 0; i < N; i += 16) {
            __m512 v = _mm512_loadu_ps(y + i);
            v = _mm512_mul_ps(v, inv);
            _mm512_storeu_ps(y + i, v);
        }
    }

    kernel_fpu_end();
}
Примечание: это упрощённая версия. В реальности нужно:

C#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/uaccess.h>
#include <linux/device.h>
#include "fwht_accel.h"
#include "fwht_avx512.h"

static dev_t devno;
static struct cdev cdev;
static struct class *cls;

static long fwht_ioctl(struct file *filp, unsigned int cmd, unsigned long arg)
{
    switch (cmd) {
    case FWHT_SINGLE: {
        struct fwht_single p;
        float buf[N];
        if (copy_from_user(&p, (void __user *)arg, sizeof(p)))
            return -EFAULT;
        if (copy_from_user(buf, (void __user *)p.in, sizeof(buf)))
            return -EFAULT;
        fwht_accel_single(buf);
        if (copy_to_user((void __user *)p.out, buf, sizeof(buf)))
            return -EFAULT;
        return 0;
    }
    case FWHT_BATCH: {
        struct fwht_batch p;
        if (copy_from_user(&p, (void __user *)arg, sizeof(p)))
            return -EFAULT;
        if (p.k == 0 || p.k > 4096)
            return -EINVAL;
        // здесь можно добавить pinned memory / hugepage проверку
        // пока просто copy → compute → copy
        size_t sz = p.k * N * sizeof(float);
        float *buf = kmalloc(sz, GFP_KERNEL);
        if (!buf) return -ENOMEM;
        if (copy_from_user(buf, (void __user *)p.in, sz)) {
            kfree(buf);
            return -EFAULT;
        }
        fwht_accel_batch(buf, p.k);
        if (copy_to_user((void __user *)p.out, buf, sz)) {
            kfree(buf);
            return -EFAULT;
        }
        kfree(buf);
        return 0;
    }
    default:
        return -ENOTTY;
    }
}

static const struct file_operations fops = {
    .owner          = THIS_MODULE,
    .unlocked_ioctl = fwht_ioctl,
    .compat_ioctl   = fwht_ioctl,
};

static int __init fwht_init(void)
{
    int ret;

    ret = alloc_chrdev_region(&devno, 0, 1, "fwht_accel");
    if (ret < 0) return ret;

    cdev_init(&cdev, &fops);
    ret = cdev_add(&cdev, devno, 1);
    if (ret < 0) goto err_chrdev;

    cls = class_create(THIS_MODULE, "fwht");
    if (IS_ERR(cls)) {
        ret = PTR_ERR(cls);
        goto err_cdev;
    }

    device_create(cls, NULL, devno, NULL, "fwht_accel");

    pr_info("fwht_accel loaded\n");
    return 0;

err_cdev:
    cdev_del(&cdev);
err_chrdev:
    unregister_chrdev_region(devno, 1);
    return ret;
}

static void __exit fwht_exit(void)
{
    device_destroy(cls, devno);
    class_destroy(cls);
    cdev_del(&cdev);
    unregister_chrdev_region(devno, 1);
    pr_info("fwht_accel unloaded\n");
}

module_init(fwht_init);
module_exit(fwht_exit);
